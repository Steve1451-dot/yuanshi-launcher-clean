cmake_minimum_required(VERSION 3.18)
project(cloudgenshin_dtls_checks VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES ON CACHE BOOL "" FORCE)
set(TEST_APPS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/extern/libpeer/third_party/mbedtls" mbedtls)
add_subdirectory("${REPO}/extern/libpeer/third_party/libsrtp" srtp)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/srtp2")
configure_file("${REPO}/extern/libpeer/third_party/libsrtp/include/srtp.h" "include/srtp2/srtp.h" COPYONLY)
set(COMMON_INCLUDES "${CMAKE_CURRENT_BINARY_DIR}/include" "${REPO}/extern/libpeer/src")
if(WIN32)
  list(PREPEND COMMON_INCLUDES "${CMAKE_CURRENT_SOURCE_DIR}/win32")
endif()
add_executable(dtls_handshake "${REPO}/tests/dtls_handshake_test.c"
  "${REPO}/extern/libpeer/src/dtls_srtp.c" "${REPO}/app/src/mbedtls_timing_compat.c")
target_include_directories(dtls_handshake PRIVATE ${COMMON_INCLUDES})
target_compile_definitions(dtls_handshake PRIVATE CONFIG_DTLS_USE_ECDSA=1 CLOUDGENSHIN_TIMING_TEST_CLOCK=1)
set_source_files_properties("${REPO}/extern/libpeer/src/dtls_srtp.c" PROPERTIES
  COMPILE_DEFINITIONS "mbedtls_entropy_init=test_entropy_init")
target_link_libraries(dtls_handshake PRIVATE mbedtls mbedx509 mbedcrypto srtp2)
if(WIN32)
  target_link_libraries(dtls_handshake PRIVATE ws2_32 bcrypt)
endif()
enable_testing()
add_test(NAME dtls_handshake COMMAND dtls_handshake)
foreach(name peer_dtls_setup peer_dtls_receive peer_dtls_loop peer_rtcp_receive)
  add_executable(${name} "${REPO}/tests/${name}_test.c" "${REPO}/extern/libpeer/src/rtcp.c" "${REPO}/extern/libpeer/src/sdp.c")
  target_include_directories(${name} PRIVATE ${COMMON_INCLUDES} "${REPO}/extern/libpeer/third_party/mbedtls/include")
  target_compile_definitions(${name} PRIVATE CONFIG_USE_USRSCTP=0)
  target_compile_options(${name} PRIVATE -ffunction-sections -fdata-sections -Werror=implicit-function-declaration -Werror=return-type)
  # PE/COFF checks undefined references before discarding unused sections.
  # Keep these existing include-the-implementation tests in one translation
  # unit so GCC can remove unrelated peer entry points before the link.
  set_target_properties(${name} PROPERTIES UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE 16)
  target_compile_options(${name} PRIVATE -O2 -fwhole-program)
  target_link_options(${name} PRIVATE -Wl,--gc-sections)
  if(WIN32)
    target_link_libraries(${name} PRIVATE ws2_32)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endforeach()

foreach(name rtcp_nack rtcp_receiver_report)
  add_executable(${name} "${REPO}/tests/${name}_test.c" "${REPO}/extern/libpeer/src/rtcp.c")
  target_include_directories(${name} PRIVATE ${COMMON_INCLUDES})
  target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -Wno-sign-compare)
  if(WIN32)
    target_compile_definitions(${name} PRIVATE __BYTE_ORDER=1234)
    target_link_libraries(${name} PRIVATE ws2_32)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endforeach()

# The Switch local-candidate fallback: gethostid() reports 0 on emulators, so
# ports.c probes the outbound IPv4 over a throwaway UDP socket. The host build
# exercises the same file through the getifaddrs branch.
add_executable(ports_host_addr "${REPO}/tests/ports_host_addr_test.c"
  "${REPO}/extern/libpeer/src/address.c")
target_include_directories(ports_host_addr PRIVATE ${COMMON_INCLUDES})
target_compile_definitions(ports_host_addr PRIVATE _WIN32_WINNT=0x0600)
target_compile_options(ports_host_addr PRIVATE -O2 -ffunction-sections -fdata-sections)
target_link_options(ports_host_addr PRIVATE -Wl,--gc-sections)
if(WIN32)
  target_link_libraries(ports_host_addr PRIVATE ws2_32 iphlpapi)
endif()
add_test(NAME ports_host_addr COMMAND ports_host_addr)
